cmake_minimum_required(VERSION 3.16)
project(CaihongSDDMConfigRegression LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 REQUIRED COMPONENTS Core)
if(NOT DEFINED SDDM_SOURCE_DIR OR NOT EXISTS "${SDDM_SOURCE_DIR}/src/common/ConfigReader.cpp")
    message(FATAL_ERROR "Set SDDM_SOURCE_DIR to the real prepared SDDM source tree")
endif()
add_executable(config-reload-regression config_reload_regression.cpp "${SDDM_SOURCE_DIR}/src/common/ConfigReader.cpp")
target_include_directories(config-reload-regression PRIVATE "${SDDM_SOURCE_DIR}/src/common")
target_link_libraries(config-reload-regression PRIVATE Qt6::Core)
enable_testing()
add_test(NAME config-reload-regression COMMAND config-reload-regression)
